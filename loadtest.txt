// Aman Bingo load test (Node 18+). Use ONLY on a TEST copy of the Worker + test D1 database.
//   1) node loadtest.mjs seed 400 > seed.sql         (makes 400 fake players with 1000 birr each)
//   2) run seed.sql on the TEST database
//   3) node loadtest.mjs run https://YOUR-TEST-WORKER.workers.dev 123456:TESTTOKEN 400
//      (TESTTOKEN must be exactly the BOT_TOKEN value set on the TEST Worker)
import crypto from 'node:crypto';
const [, , mode, a1, a2, a3] = process.argv;
const BASE_ID = 900000001, START_BIRR = 1000, uid = i => BASE_ID + i, sleep = ms => new Promise(r => setTimeout(r, ms));

if (mode === 'seed') {
  const n = +a1 || 400, now = Date.now();
  console.log(`DELETE FROM users WHERE id>=${uid(0)} AND id<=${uid(n - 1)};`);
  for (let i = 0; i < n; i++) console.log(`INSERT INTO users(id,name,phone,balance,bonus_c,bonus_period,created_ts) VALUES(${uid(i)},'Test${i}','09${10000000 + i}',${START_BIRR * 100},0,999999,${now});`);
  process.exit(0);
}
if (mode !== 'run' || !a1 || !a2) { console.log('Usage: see the top of this file'); process.exit(1) }

const BASE = a1.replace(/\/$/, ''), TOKEN = a2, N = +a3 || 400;
const initData = id => {
  const p = { auth_date: String(Math.floor(Date.now() / 1000)), user: JSON.stringify({ id, first_name: 'Test' + id }) };
  const str = Object.keys(p).sort().map(k => k + '=' + p[k]).join('\n');
  const secret = crypto.createHmac('sha256', 'WebAppData').update(TOKEN).digest();
  return new URLSearchParams({ ...p, hash: crypto.createHmac('sha256', secret).update(str).digest('hex') }).toString();
};
const INIT = Array.from({ length: N }, (_, i) => initData(uid(i)));
const lat = { pick: [], state: [], me: [] }, errs = {};
const bump = k => errs[k] = (errs[k] || 0) + 1;
async function call(i, path, body, kind) {
  const t = performance.now();
  try {
    const r = await fetch(BASE + path, { method: body ? 'POST' : 'GET', headers: { 'content-type': 'application/json', 'x-init-data': INIT[i] }, body: body ? JSON.stringify(body) : undefined, signal: AbortSignal.timeout(15000) });
    const d = await r.json().catch(() => ({})); lat[kind].push(performance.now() - t);
    if (!r.ok) { bump(kind + ':' + (d.error || 'http_' + r.status)); return { ok: false, error: d.error } }
    return { ok: true, d };
  } catch (e) { lat[kind].push(performance.now() - t); bump(kind + ':' + (e.name || 'network')); return { ok: false, error: 'net' } }
}
const pct = (a, p) => a.length ? [...a].sort((x, y) => x - y)[Math.min(a.length - 1, Math.floor(a.length * p))].toFixed(0) : '-';
const ids = [...Array(N).keys()];

console.log(`Waiting for a fresh lobby with 25+ seconds left...`);
let rid, stake;
for (;;) {
  const r = await call(0, '/api/state', null, 'state');
  if (!r.ok) { console.log('Cannot reach the game:', r.error, '- check URL, BOT_TOKEN and that seed.sql was run'); process.exit(1) }
  const s = r.d, off = s.now - Date.now();
  if (s.round.status === 'lobby' && s.round.start_ts + s.round.lobby_ms - (Date.now() + off) > 25000) { rid = s.round.id; stake = s.round.stake; break }
  await sleep(1000);
}
console.log(`Round ${rid}, stake ${stake} birr. PHASE A: ${N} players grab cartela #1 at the same instant...`);
const A = await Promise.all(ids.map(i => call(i, '/api/pick', { cartela: 1 }, 'pick')));
const raceWins = A.filter(x => x.ok).length;
console.log(`PHASE B: every player picks their own cartela...`);
const perm = [...Array(399).keys()].map(x => x + 2).sort(() => Math.random() - .5);
const B = await Promise.all(ids.map(i => call(i, '/api/pick', { cartela: perm[i % 399] }, 'pick')));
const cards = raceWins + B.filter(x => x.ok).length;
const chk = await call(0, '/api/state', null, 'state'), taken = chk.d.taken;
console.log(`Picks accepted: ${cards}. Now ${N} players poll like real players until the round ends...`);

let ended = null; const deadline = Date.now() + 8 * 60 * 1000;
await Promise.all(ids.map(async i => {
  await sleep(Math.random() * 2000);
  while (!ended && Date.now() < deadline) {
    const r = await call(i, '/api/state', null, 'state');
    if (r.ok && r.d.round.id === rid && (r.d.round.status === 'ended' || r.d.round.status === 'cancelled')) { ended = r.d; break }
    await sleep(2000 + Math.random() * 1000);
  }
}));
if (!ended) { console.log('FAIL: the round never finished within 8 minutes'); process.exit(1) }

let total = 0; await Promise.all(ids.map(async i => { const r = await call(i, '/api/me', null, 'me'); if (r.ok) total += Math.round((r.d.balance + r.d.bonus) * 100) }));
const stakeC = Math.round(stake * 100), start = N * START_BIRR * 100;
let expected = start;
if (ended.round.status === 'ended') { const prize = Math.floor(cards * stakeC * 0.8), w = ended.winners.length; expected = start - cards * stakeC + Math.floor(prize / w) * w }
const unexpected = Object.entries(errs).filter(([k]) => k !== 'pick:taken');
const res = [
  ['Only ONE player got cartela #1 (race test)', raceWins === 1, `${raceWins} got it`],
  ['No cartela handed out twice', taken.length === cards && new Set(taken).size === taken.length, `${cards} accepted, ${taken.length} taken, ${new Set(taken).size} unique`],
  ['Money adds up after the round', total === expected, `expected ${expected / 100}, got ${total / 100} birr in total`],
  ['Pick speed p95 under 1.5 s', +pct(lat.pick, .95) < 1500, `p95 ${pct(lat.pick, .95)} ms`],
  ['State speed p95 under 1 s', +pct(lat.state, .95) < 1000, `p95 ${pct(lat.state, .95)} ms`],
  ['No unexpected errors', unexpected.length === 0, unexpected.map(([k, v]) => k + '=' + v).join(', ') || 'none']];
console.log(`\nRound ended as: ${ended.round.status}, winners: ${ended.winners.map(x => x.name).join(', ') || '-'}`);
console.log(`Latency ms   pick p50/p95/max: ${pct(lat.pick, .5)}/${pct(lat.pick, .95)}/${pct(lat.pick, 1)}   state p50/p95/max: ${pct(lat.state, .5)}/${pct(lat.state, .95)}/${pct(lat.state, 1)}   requests: ${lat.pick.length + lat.state.length + lat.me.length}`);
console.log(`Expected 'taken' errors (players fighting for the same cartela): ${errs['pick:taken'] || 0}\n`);
for (const [n, ok, d] of res) console.log((ok ? 'PASS  ' : 'FAIL  ') + n + '  (' + d + ')');
console.log(res.every(r => r[1]) ? '\nALL PASSED' : '\nSOME TESTS FAILED - send me this output');
